Events

<script>
  const ta = document.getElementById('my-textarea');
  ta.addEventListener('native:input', (e) => {
    console.log('input:', e.detail.value);
  });
  ta.addEventListener('native:change', (e) => {
    console.log('change:', e.detail.value);
  });
</script>
