Usage

/* ── Color Environment Parameters ── */
:root {
  --n-env-lightness-min: 0.25;
  --n-env-lightness-max: 0.98;
  --n-env-chroma: 0.15;
  --n-env-alpha: 0.5;
}

/* ── Per-Family Hue Override ── */
:root {
  --n-env-hue-accent: 250;
  --n-env-hue-success: 145;
  --n-env-hue-danger: 25;
}

/* ── Using Semantic Tokens ── */
.card {
  background: var(--n-panel-accent);
  color: var(--n-ink-accent);
  border: 1px solid var(--n-border-muted-accent);
}

.surface-fill {
  background: var(--n-surface-accent);
  color: var(--n-surface-ink-accent);
}
