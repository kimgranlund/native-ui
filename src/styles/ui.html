Cascading

Parent intent="accent", child overrides

Section cascading all attributes

<!-- Attributes cascade from parent to children -->
<section size="sm" density="compact" radius="round" intent="accent">
  <n-button variant="primary">Inherits all</n-button>
  <n-button variant="secondary">Inherits all</n-button>

  <!-- Child can override any attribute -->
  <n-button variant="primary" intent="danger">Override intent</n-button>
  <n-button variant="primary" size="lg">Override size</n-button>
</section>
