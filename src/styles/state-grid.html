/* ── State Token Resolution ── */
/* Components set explicit defaults; attribute selectors override */

/* Rest state */
background: var(--n-background);
color: var(--n-color);
border-color: var(--n-border-color);

/* Hover state */
background: var(--n-background-hover);
color: var(--n-color-hover);
border-color: var(--n-border-color-hover);

/* Disabled state */
color: var(--n-ink-disabled);
pointer-events: none;
/* ── Intent → Variant Resolution Flow ── */
/* 1. [intent] sets the color family */
:where([intent="accent"]) {
  --n-panel: var(--n-panel-accent);
  --n-surface: var(--n-surface-accent);
  --n-ink: var(--n-ink-accent);
}

/* 2. [variant] picks the role from the family */
:where([variant="primary"]) {
  --n-background: var(--n-surface);
  --n-color: var(--n-surface-ink);
  --n-border-color: transparent;
}
